{{- /* A card that shows multiple images in a single card, then links to one page. Built on top of Hextra's card markup so it matches the theme styling. @param {string} link The link the card points to. @param {string} title The title shown under the images. @param {string} subtitle Optional subtitle text. @inner One image URL per line (local or remote). @example {{< cards >}} {{< card-gallery link="effects/foo/" title="Speed lines" >}} https://example.com/a.gif https://example.com/b.gif {{< /card-gallery >}} {{< /cards >}} */ -}} {{- $link := .Get "link" -}} {{- $title := .Get "title" -}} {{- $subtitle := .Get "subtitle" -}} {{- $images := slice -}} {{- range (split (.Inner | default "") "\n") -}} {{- with (trim . " \t\r") -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- $count := len $images -}} {{- $width := cond (gt $count 0) (printf "%d%%" (div 100 $count)) "100%" -}} {{- $external := strings.HasPrefix $link "http" -}} {{- $href := cond (strings.HasPrefix $link "/") ($link | relURL) $link -}}
{{- range $images -}} {{- end -}}
{{ $title }} {{- with $subtitle -}}
{{ . | markdownify }}
{{- end -}}
